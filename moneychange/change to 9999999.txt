using BepInEx;
using BepInEx.Logging;
using HarmonyLib;
using UnityEngine;
using System.Linq;
using System.Reflection;

namespace FishUnlimitedMoneyFinal
{
    [BepInPlugin("me.yexiao.fish.finalmod", "FishUnlimitedMoneyFinal", "1.7.1")]
    public class FishUnlimitedMoneyFinal : BaseUnityPlugin
    {
        public static int CustomMoneyValue = 9999999;
        private Harmony _harmony;
        private static object _moneyUITextObj;
        private static MethodInfo _setTextMethod;
        private static FieldInfo _underlyingMoneyField;

        void Awake()
        {
            Logger.LogInfo("✅ 终极锁死版无限货币Mod加载完成，底层数值完全锁定无跳动");
            _harmony = new Harmony("me.yexiao.fish.finalmod");
            
            var moneyMgrType = System.Type.GetType("MoneyManager, Assembly-CSharp");

            // 1. 拦截Money属性get，永远返回9999999
            var moneyGetterPatch = new HarmonyMethod(typeof(Patch_MoneyGetter).GetMethod("Prefix"));
            var moneyGetter = AccessTools.PropertyGetter(moneyMgrType, "Money");
            _harmony.Patch(moneyGetter, prefix: moneyGetterPatch);

            // 2. 拦截Money属性set，完全禁止游戏修改货币
            var moneySetterPatch = new HarmonyMethod(typeof(Patch_MoneySetter).GetMethod("Prefix"));
            var moneySetter = AccessTools.PropertySetter(moneyMgrType, "Money");
            _harmony.Patch(moneySetter, prefix: moneySetterPatch);

            // 3. 拦截所有消费判断方法，永远返回true
            var allCheckMethods = moneyMgrType.GetMethods(BindingFlags.Public | BindingFlags.NonPublic | BindingFlags.Instance)
                .Where(method => 
                    method.ReturnType == typeof(bool) && 
                    (method.Name.ToLower().Contains("afford") || method.Name.ToLower().Contains("check") || method.Name.ToLower().Contains("canbuy"))
                );
            foreach (var method in allCheckMethods)
            {
                _harmony.Patch(method, prefix: new HarmonyMethod(typeof(Patch_CheckAfford).GetMethod("Prefix")));
            }

            // 4. 用标准反射直接找到MoneyManager里存储货币的私有int字段，替代不存在的FirstField
            _underlyingMoneyField = moneyMgrType.GetFields(BindingFlags.Public | BindingFlags.NonPublic | BindingFlags.Instance)
                .FirstOrDefault(f => f.FieldType == typeof(int));
        }

        void Update()
        {
            // 小键盘1：货币 +9999999
            if (Input.GetKeyDown(KeyCode.Keypad1))
            {
                CustomMoneyValue += 9999999;
                Logger.LogInfo($"✅ 按下小键盘1，当前货币已更新为：{CustomMoneyValue}");
            }

            // 小键盘2：一键重置货币为9999999
            if (Input.GetKeyDown(KeyCode.Keypad2))
            {
                CustomMoneyValue = 9999999;
                Logger.LogInfo($"✅ 按下小键盘2，货币已重置为9999999");
            }

            // 兜底强制锁死底层私有货币字段，游戏任何修改都立刻被覆盖
            if (_underlyingMoneyField != null)
            {
                // 替换过时API，用Unity新版的FindAnyObjectByType查找MoneyManager实例
                var moneyMgrInstance = Object.FindAnyObjectByType(_underlyingMoneyField.DeclaringType);
                if (moneyMgrInstance != null)
                {
                    _underlyingMoneyField.SetValue(moneyMgrInstance, CustomMoneyValue);
                }
            }

            // 纯反射自动适配所有UI文本类型，不需要任何引用
            if (_moneyUITextObj == null)
            {
                // 先尝试找TextMeshPro类型
                var tmpType = AccessTools.TypeByName("TMPro.TextMeshProUGUI");
                if (tmpType != null)
                {
                    var allTexts = Resources.FindObjectsOfTypeAll(tmpType);
                    _moneyUITextObj = allTexts.FirstOrDefault(t => 
                        (string)tmpType.GetProperty("text").GetValue(t) != null && 
                        ((string)tmpType.GetProperty("text").GetValue(t)).All(char.IsDigit)
                    );
                    if (_moneyUITextObj != null)
                    {
                        _setTextMethod = tmpType.GetProperty("text").GetSetMethod();
                    }
                }

                // 找不到就回退找旧版UGUI Text
                if (_moneyUITextObj == null)
                {
                    var uiTextType = AccessTools.TypeByName("UnityEngine.UI.Text");
                    if (uiTextType != null)
                    {
                        var allTexts = Resources.FindObjectsOfTypeAll(uiTextType);
                        _moneyUITextObj = allTexts.FirstOrDefault(t => 
                            (string)uiTextType.GetProperty("text").GetValue(t) != null && 
                            ((string)uiTextType.GetProperty("text").GetValue(t)).All(char.IsDigit)
                        );
                        if (_moneyUITextObj != null)
                        {
                            _setTextMethod = uiTextType.GetProperty("text").GetSetMethod();
                        }
                    }
                }
            }

            // 每帧强制刷新货币UI
            if (_moneyUITextObj != null && _setTextMethod != null)
            {
                _setTextMethod.Invoke(_moneyUITextObj, new object[] { CustomMoneyValue.ToString() });
            }
        }

        void OnDestroy()
        {
            _harmony.UnpatchSelf();
        }
    }

    public static class Patch_MoneyGetter
    {
        public static bool Prefix(ref int __result)
        {
            __result = FishUnlimitedMoneyFinal.CustomMoneyValue;
            return false;
        }
    }

    public static class Patch_MoneySetter
    {
        public static bool Prefix()
        {
            return false;
        }
    }

    public static class Patch_CheckAfford
    {
        public static bool Prefix(ref bool __result)
        {
            __result = true;
            return false;
        }
    }
}
